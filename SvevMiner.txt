import React, { useState, useEffect, useRef, useMemo } from 'react';
import { initializeApp } from 'firebase/app';
import { 
  getAuth, 
  signInAnonymously, 
  signInWithCustomToken,
  onAuthStateChanged 
} from 'firebase/auth';
import { 
  getFirestore, 
  collection, 
  doc, 
  setDoc, 
  onSnapshot, 
  serverTimestamp, 
  addDoc
} from 'firebase/firestore';
import { Mic, MicOff, MessageCircle, X, Send, Zap, Pickaxe } from 'lucide-react';

// --- Firebase Setup ---
const firebaseConfig = JSON.parse(__firebase_config);
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';

// --- Assets & Helpers ---
const getRandomColor = () => {
  const hues = [190, 220, 280, 320, 150]; // Cyan, Blue, Purple, Pink, Green
  return hues[Math.floor(Math.random() * hues.length)];
};

const getRandomName = () => {
  const adjs = ['Snill', 'Hjelpsom', 'Kosmisk', 'Svevende', 'Glødende', 'Rask'];
  const nouns = ['Pilot', 'Miner', 'Vandrer', 'Stjerne', 'Drømmer', 'Navigatør'];
  return `${adjs[Math.floor(Math.random() * adjs.length)]} ${nouns[Math.floor(Math.random() * nouns.length)]}`;
};

class Particle {
  constructor(x, y, color, speed) {
    this.x = x;
    this.y = y;
    const angle = Math.random() * Math.PI * 2;
    this.vx = Math.cos(angle) * speed;
    this.vy = Math.sin(angle) * speed;
    this.life = 1.0;
    this.decay = Math.random() * 0.02 + 0.01;
    this.color = color;
  }
  update() {
    this.x += this.vx;
    this.y += this.vy;
    this.life -= this.decay;
  }
  draw(ctx, offsetX, offsetY) {
    ctx.globalAlpha = this.life;
    ctx.fillStyle = this.color;
    ctx.beginPath();
    ctx.arc(this.x + offsetX, this.y + offsetY, Math.random() * 3, 0, Math.PI * 2);
    ctx.fill();
    ctx.globalAlpha = 1.0;
  }
}

export default function SvevMining() {
  const [user, setUser] = useState(null);
  const [messages, setMessages] = useState([]);
  const [showGlobalChat, setShowGlobalChat] = useState(false);
  const [inputText, setInputText] = useState("");
  const [isMicOn, setIsMicOn] = useState(false);
  const [scoreDisplay, setScoreDisplay] = useState(0);
  const [otherPlayers, setOtherPlayers] = useState({});
  const [nearbyCount, setNearbyCount] = useState(0);
  const [miningTarget, setMiningTarget] = useState(null); // ID of rock being mined

  const canvasRef = useRef(null);
  const requestRef = useRef();
  
  // Game State
  const gameState = useRef({
    x: 1500, y: 1500, vx: 0, vy: 0,
    angle: 0,
    colorHue: getRandomColor(),
    name: getRandomName(),
    score: 0,
    particles: [],
    shake: 0,
    laserIntensity: 0
  });

  const rocksRef = useRef([]); 
  const joystick = useRef({ active: false, dx: 0, dy: 0, originX: 0, originY: 0, currX: 0, currY: 0 });
  const WORLD_SIZE = 4000;

  // --- Initialization ---
  useEffect(() => {
    // Generate Rocks (Client side sync simulation for prototype)
    // In a real app, rock positions/HP would be synced in Firestore
    rocksRef.current = Array.from({ length: 25 }).map((_, i) => ({
      id: i,
      x: Math.random() * WORLD_SIZE,
      y: Math.random() * WORLD_SIZE,
      size: Math.random() * 60 + 40, // Big rocks
      hp: 100,
      maxHp: 100,
      type: Math.random() > 0.8 ? 'RARE' : 'COMMON',
      rotation: Math.random() * Math.PI,
      rotSpeed: (Math.random() - 0.5) * 0.01
    }));

    const initAuth = async () => {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
            await signInWithCustomToken(auth, __initial_auth_token);
        } else {
            await signInAnonymously(auth);
        }
    };
    initAuth();

    const unsubscribe = onAuthStateChanged(auth, u => { if (u) setUser(u); });
    return () => unsubscribe();
  }, []);

  // --- Multiplayer Sync ---
  useEffect(() => {
    if (!user) return;
    
    // Players
    const unsubPlayers = onSnapshot(collection(db, 'artifacts', appId, 'public', 'data', 'svev_players'), snap => {
      const p = {};
      snap.forEach(d => { if(d.id !== user.uid) p[d.id] = d.data(); });
      setOtherPlayers(p);
    });

    // Chat
    const unsubChat = onSnapshot(collection(db, 'artifacts', appId, 'public', 'data', 'svev_chat'), snap => {
      const m = [];
      snap.forEach(d => m.push({id: d.id, ...d.data()}));
      m.sort((a,b) => (a.createdAt?.seconds || 0) - (b.createdAt?.seconds || 0));
      setMessages(m.slice(-50));
    });

    // Broadcast self
    const interval = setInterval(() => {
      setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'svev_players', user.uid), {
        x: gameState.current.x,
        y: gameState.current.y,
        colorHue: gameState.current.colorHue,
        name: gameState.current.name,
        mining: gameState.current.laserIntensity > 0, // Broadcast that I am mining
        lastSeen: serverTimestamp()
      }, { merge: true });
    }, 500);

    return () => { unsubPlayers(); unsubChat(); clearInterval(interval); };
  }, [user]);


  // --- GAME LOOP ---
  const animate = () => {
    if (!canvasRef.current) return;
    const canvas = canvasRef.current;
    const ctx = canvas.getContext('2d');
    const state = gameState.current;

    // 1. Movement Physics
    const acceleration = 0.5;
    const friction = 0.94;
    
    if (joystick.current.active) {
      state.vx += joystick.current.dx * acceleration;
      state.vy += joystick.current.dy * acceleration;
      state.angle = Math.atan2(joystick.current.dy, joystick.current.dx);
    }
    state.vx *= friction;
    state.vy *= friction;
    state.x = Math.max(0, Math.min(state.x + state.vx, WORLD_SIZE));
    state.y = Math.max(0, Math.min(state.y + state.vy, WORLD_SIZE));

    // 2. Camera Setup
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
    
    // Screen Shake
    if (state.shake > 0) state.shake *= 0.9;
    const shakeX = (Math.random() - 0.5) * state.shake;
    const shakeY = (Math.random() - 0.5) * state.shake;
    
    const centerX = canvas.width / 2;
    const centerY = canvas.height / 2;
    const offsetX = centerX - state.x + shakeX;
    const offsetY = centerY - state.y + shakeY;

    // 3. Render Background
    const bgGrad = ctx.createLinearGradient(0, 0, 0, canvas.height);
    bgGrad.addColorStop(0, '#0f172a');
    bgGrad.addColorStop(1, '#1e1b4b');
    ctx.fillStyle = bgGrad;
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    // Grid
    ctx.strokeStyle = 'rgba(255,255,255,0.03)';
    ctx.lineWidth = 1;
    const gridSize = 150;
    const startX = Math.floor((state.x - centerX) / gridSize) * gridSize;
    const startY = Math.floor((state.y - centerY) / gridSize) * gridSize;
    for (let x = startX - gridSize; x < startX + canvas.width + gridSize; x += gridSize) {
        ctx.beginPath(); ctx.moveTo(x + offsetX, -WORLD_SIZE); ctx.lineTo(x + offsetX, WORLD_SIZE * 2); ctx.stroke();
    }
    for (let y = startY - gridSize; y < startY + canvas.height + gridSize; y += gridSize) {
        ctx.beginPath(); ctx.moveTo(-WORLD_SIZE, y + offsetY); ctx.lineTo(WORLD_SIZE * 2, y + offsetY); ctx.stroke();
    }

    // 4. Mining Logic & Rocks
    let activeTargetId = null;
    let closestDist = 9999;

    rocksRef.current.forEach(rock => {
       // Rock visuals
       const dist = Math.hypot(state.x - rock.x, state.y - rock.y);
       
       // Culling
       if (dist > 1000) return;

       rock.rotation += rock.rotSpeed;

       // Draw Rock
       ctx.save();
       ctx.translate(rock.x + offsetX, rock.y + offsetY);
       ctx.rotate(rock.rotation);
       
       ctx.fillStyle = rock.type === 'RARE' ? '#4c1d95' : '#334155';
       ctx.strokeStyle = rock.type === 'RARE' ? '#a78bfa' : '#64748b';
       ctx.lineWidth = 3;
       
       // Draw jagged polygon
       ctx.beginPath();
       const sides = 6;
       for(let i=0; i<sides; i++) {
           const ang = (i / sides) * Math.PI * 2;
           const r = rock.size * (1 + Math.sin(ang * 3 + rock.id) * 0.1); // Jaggedness
           ctx.lineTo(Math.cos(ang) * r, Math.sin(ang) * r);
       }
       ctx.closePath();
       ctx.fill();
       ctx.stroke();

       // Health Bar
       if (rock.hp < rock.maxHp) {
           ctx.fillStyle = '#000';
           ctx.fillRect(-30, -rock.size - 20, 60, 6);
           ctx.fillStyle = '#10b981';
           ctx.fillRect(-29, -rock.size - 19, 58 * (rock.hp / rock.maxHp), 4);
       }
       ctx.restore();

       // Mining Check (Auto-mine if close and stopped moving)
       const mineRange = 250;
       if (dist < mineRange) {
           if (dist < closestDist) {
               closestDist = dist;
               activeTargetId = rock.id;
           }
       }
    });

    // Handle Mining Beam
    if (activeTargetId !== null && !joystick.current.active) {
        setMiningTarget(activeTargetId);
        state.laserIntensity = Math.min(state.laserIntensity + 0.1, 1);
        
        const rock = rocksRef.current.find(r => r.id === activeTargetId);
        if (rock) {
            // Damage rock
            rock.hp -= 0.5;
            state.score += 0.1;
            setScoreDisplay(Math.floor(state.score));
            state.shake = 1; // Mild shake

            // Beam Visuals
            ctx.beginPath();
            ctx.moveTo(centerX, centerY);
            ctx.lineTo(rock.x + offsetX, rock.y + offsetY);
            ctx.strokeStyle = `rgba(${state.type === 'RARE' ? '200,100,255' : '100,255,255'}, ${0.5 + Math.random()*0.5})`;
            ctx.lineWidth = 4 + Math.random() * 4;
            ctx.stroke();

            // Sparks at impact
            if (Math.random() > 0.5) {
                state.particles.push(new Particle(rock.x, rock.y, '#fff', 3));
            }

            // Rock Destroyed?
            if (rock.hp <= 0) {
                // Explosion
                for(let i=0; i<20; i++) state.particles.push(new Particle(rock.x, rock.y, '#fbbf24', 6));
                state.shake = 10;
                
                // Respawn elsewhere
                rock.x = Math.random() * WORLD_SIZE;
                rock.y = Math.random() * WORLD_SIZE;
                rock.hp = rock.maxHp;
            }
        }
    } else {
        setMiningTarget(null);
        state.laserIntensity = 0;
    }


    // 5. Other Players
    Object.values(otherPlayers).forEach(p => {
        const pX = p.x + offsetX;
        const pY = p.y + offsetY;
        
        // Skip if offscreen
        if (pX < -50 || pX > canvas.width+50) return;

        // Player Body
        ctx.beginPath();
        ctx.arc(pX, pY, 15, 0, Math.PI*2);
        ctx.fillStyle = `hsl(${p.colorHue}, 70%, 50%)`;
        ctx.fill();

        // Name
        ctx.fillStyle = 'white';
        ctx.font = '10px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText(p.name, pX, pY - 25);

        // Their Laser Beam (Visual only based on their state)
        if (p.mining) {
             // Find nearest rock to them visually to connect beam
             // (In real app, we would sync targetId)
             let nearestRock = null;
             let minDist = 300;
             rocksRef.current.forEach(r => {
                 const d = Math.hypot(p.x - r.x, p.y - r.y);
                 if (d < minDist) { minDist = d; nearestRock = r; }
             });

             if (nearestRock) {
                 ctx.beginPath();
                 ctx.moveTo(pX, pY);
                 ctx.lineTo(nearestRock.x + offsetX, nearestRock.y + offsetY);
                 ctx.strokeStyle = `hsla(${p.colorHue}, 100%, 70%, 0.5)`;
                 ctx.lineWidth = 2;
                 ctx.stroke();
             }
        }
    });

    // 6. Draw Self
    ctx.beginPath();
    ctx.arc(centerX, centerY, 15, 0, Math.PI*2);
    ctx.fillStyle = `hsl(${state.colorHue}, 100%, 50%)`;
    ctx.shadowBlur = 15;
    ctx.shadowColor = `hsl(${state.colorHue}, 100%, 50%)`;
    ctx.fill();
    ctx.shadowBlur = 0;
    
    // Joystick Visual
    if (joystick.current.active) {
        ctx.beginPath();
        ctx.arc(joystick.current.currX, joystick.current.currY, 25, 0, Math.PI*2);
        ctx.fillStyle = 'rgba(255,255,255,0.2)';
        ctx.fill();
    }

    // Particles
    state.particles.forEach(p => { p.update(); p.draw(ctx, offsetX, offsetY); });
    state.particles = state.particles.filter(p => p.life > 0);

    requestRef.current = requestAnimationFrame(animate);
  };

  useEffect(() => {
    requestRef.current = requestAnimationFrame(animate);
    return () => cancelAnimationFrame(requestRef.current);
  }, [otherPlayers]);

  // --- Input ---
  const handleInput = (type, x, y) => {
      if (type === 'start') {
          joystick.current.active = true;
          joystick.current.originX = x; joystick.current.originY = y;
          joystick.current.currX = x; joystick.current.currY = y;
      } else if (type === 'move' && joystick.current.active) {
          const dx = x - joystick.current.originX;
          const dy = y - joystick.current.originY;
          const dist = Math.hypot(dx, dy);
          const max = 50;
          const clamp = Math.min(dist, max);
          const angle = Math.atan2(dy, dx);
          
          joystick.current.currX = joystick.current.originX + Math.cos(angle) * clamp;
          joystick.current.currY = joystick.current.originY + Math.sin(angle) * clamp;
          joystick.current.dx = Math.cos(angle) * (clamp/max);
          joystick.current.dy = Math.sin(angle) * (clamp/max);
      } else if (type === 'end') {
          joystick.current.active = false;
          joystick.current.dx = 0; joystick.current.dy = 0;
      }
  };

  const sendMessage = async (e) => {
    e.preventDefault();
    if (!inputText.trim()) return;
    await addDoc(collection(db, 'artifacts', appId, 'public', 'data', 'svev_chat'), {
      text: inputText, senderId: user.uid, senderName: gameState.current.name, createdAt: serverTimestamp()
    });
    setInputText("");
  };

  if (!user) return <div className="h-screen bg-slate-900 flex items-center justify-center text-cyan-400 animate-pulse">Laster Mining-Laser...</div>;

  return (
    <div className="fixed inset-0 overflow-hidden bg-slate-900 select-none font-sans">
        <canvas 
            ref={canvasRef}
            className="absolute inset-0"
            onTouchStart={e => handleInput('start', e.touches[0].clientX, e.touches[0].clientY)}
            onTouchMove={e => handleInput('move', e.touches[0].clientX, e.touches[0].clientY)}
            onTouchEnd={() => handleInput('end')}
            onMouseDown={e => handleInput('start', e.clientX, e.clientY)}
            onMouseMove={e => handleInput('move', e.clientX, e.clientY)}
            onMouseUp={() => handleInput('end')}
        />

        {/* HUD */}
        <div className="absolute top-4 left-4 pointer-events-none">
            <h1 className="text-2xl font-black italic text-cyan-400 drop-shadow-lg tracking-tighter">SVEV <span className="text-white text-sm font-normal not-italic ml-2">MINER</span></h1>
            <div className="text-xs text-slate-400 mt-1 flex gap-2">
                <span className="bg-slate-800 px-2 py-1 rounded border border-slate-700">RESSURSER: {scoreDisplay}</span>
            </div>
        </div>

        {/* Mining Indicator */}
        {miningTarget !== null && (
            <div className="absolute top-20 left-1/2 -translate-x-1/2 text-center pointer-events-none">
                <div className="bg-yellow-500/20 backdrop-blur border border-yellow-500/50 text-yellow-300 px-4 py-1 rounded-full text-xs font-bold animate-pulse flex items-center gap-2">
                    <Pickaxe size={12} />
                    MINING IN PROGRESS
                </div>
                <div className="text-[10px] text-white/50 mt-1">Stopp bevegelse for å mine</div>
            </div>
        )}

        {/* Action Buttons */}
        <div className="absolute top-4 right-4 flex gap-3 pointer-events-auto">
             <button onClick={() => setShowGlobalChat(!showGlobalChat)} className="bg-slate-800/80 p-3 rounded-full text-white border border-white/10 backdrop-blur">
                <MessageCircle size={20} />
             </button>
        </div>

        {/* Proximity / Social Cues */}
        <div className="absolute bottom-6 left-6 right-6 flex items-end justify-between pointer-events-none">
            <div className="pointer-events-auto">
                 <button 
                    onClick={() => setIsMicOn(!isMicOn)}
                    className={`w-12 h-12 rounded-full flex items-center justify-center border border-white/10 transition-all ${isMicOn ? 'bg-green-500 text-black' : 'bg-slate-800/80 text-slate-400'}`}
                 >
                    {isMicOn ? <Mic size={20} /> : <MicOff size={20} />}
                 </button>
            </div>
            
            {miningTarget !== null && (
                <div className="text-right pointer-events-none">
                    <div className="text-xs text-cyan-300 font-bold mb-1">SIGNAL BOOST</div>
                    <div className="text-[10px] text-slate-400 max-w-[150px]">
                        Bruk chatten til å be andre hjelpe deg med store steiner!
                    </div>
                </div>
            )}
        </div>

        {/* Chat Overlay */}
        {showGlobalChat && (
            <div className="absolute inset-0 bg-black/80 z-50 flex flex-col pointer-events-auto p-4 md:p-10">
                <div className="flex-1 bg-slate-900 rounded-2xl border border-white/10 overflow-hidden flex flex-col max-w-lg mx-auto w-full">
                    <div className="p-4 border-b border-white/10 flex justify-between items-center bg-slate-800">
                        <span className="font-bold text-white text-sm">Sektor Chat</span>
                        <button onClick={() => setShowGlobalChat(false)}><X size={18} className="text-white"/></button>
                    </div>
                    <div className="flex-1 overflow-y-auto p-4 space-y-3">
                         {messages.map(m => (
                             <div key={m.id} className="text-sm">
                                 <span className="text-cyan-400 font-bold text-xs block">{m.senderName}</span>
                                 <span className="text-slate-300">{m.text}</span>
                             </div>
                         ))}
                    </div>
                    <form onSubmit={sendMessage} className="p-3 bg-black/20 flex gap-2">
                        <input value={inputText} onChange={e=>setInputText(e.target.value)} className="flex-1 bg-slate-800 rounded-full px-4 text-white text-sm outline-none border border-transparent focus:border-cyan-500" placeholder="Skriv melding..." />
                        <button className="bg-cyan-600 text-white p-2 rounded-full"><Send size={16}/></button>
                    </form>
                </div>
            </div>
        )}
    </div>
  );
}
